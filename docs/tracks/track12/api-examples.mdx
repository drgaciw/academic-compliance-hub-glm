# API Code Examples

This guide provides code examples for all major tRPC procedures and API endpoints.

## Authentication

### Login

```typescript
import { trpc } from "@packages/trpc/client";

const login = async (email: string, password: string) => {
  const result = await trpc.auth.login.mutate({
    email,
    password,
  });

  // Store token
  localStorage.setItem("token", result.token);

  return result;
};

// Usage
const user = await login("student@example.com", "password123");
```

### Logout

```typescript
const logout = async () => {
  await trpc.auth.logout.mutate();
  localStorage.removeItem("token");
};

// Usage
await logout();
```

### Get Current User

```typescript
const getCurrentUser = async () => {
  const user = await trpc.auth.getCurrentUser.query();
  return user;
};

// Usage
const currentUser = await getCurrentUser();
console.log(currentUser.name);
```

## File Uploads

### Upload Document

```typescript
const uploadDocument = async (file: File, type: "transcript" | "form") => {
  const formData = new FormData();
  formData.append("file", file);
  formData.append("type", type);

  const result = await trpc.documents.upload.mutate(formData);
  return result;
};

// Usage
const fileInput = document.querySelector('input[type="file"]');
const file = fileInput.files[0];
const uploaded = await uploadDocument(file, "transcript");
```

### Get Document

```typescript
// Usage

```

## Error Handling

### Try-Catch Pattern

```typescript
try {
  const user = await trpc.users.byId.query({ id: "user-123" });
  console.log(user);
} catch (error) {
  if (error instanceof TRPCError) {
    console.error("Error:", error.message);
    console.error("Code:", error.code);

    if (error.code === "NOT_FOUND") {
      console.log("User not found");
    } else if (error.code === "UNAUTHORIZED") {
      console.log("Not authorized");
    }
  }
}
```

### Loading States

```typescript
const { data, isLoading, error } = trpc.users.byId.useQuery(
  { id: 'user-123' },
  {
    onSuccess: (data) => {
      console.log('User loaded:', data);
    },
    onError: (error) => {
      console.error('Failed to load user:', error);
    },
  }
);

if (isLoading) return <div>Loading...</div>;
if (error) return <div>Error: {error.message}</div>;

return <div>{data?.name}</div>;
```
